import atheris
import sys
import torch


def _pick_dtype(fdp):
    return fdp.PickValueInList([
        torch.float32,
        torch.float64,
        torch.int32,
        torch.int64,
        torch.bool,
    ])


def _shape_numel(shape):
    n = 1
    for s in shape:
        n *= int(s)
    return n


def _consume_shape(fdp, rank, min_dim=0, max_dim=4):
    return [fdp.ConsumeIntInRange(min_dim, max_dim) for _ in range(rank)]


def _make_tensor(fdp, shape, dtype):
    numel = _shape_numel(shape)
    if dtype == torch.bool:
        vals = [fdp.ConsumeBool() for _ in range(numel)]
        return torch.tensor(vals, dtype=dtype).reshape(shape)
    elif dtype.is_floating_point:
        vals = []
        for _ in range(numel):
            if fdp.ConsumeBool():
                vals.append(fdp.ConsumeFloatInRange(-10.0, 10.0))
            else:
                vals.append(float(fdp.PickValueInList([0.0, -0.0, 1.0, -1.0, 2.5, -3.5])))
        return torch.tensor(vals, dtype=dtype).reshape(shape)
    else:
        vals = [fdp.ConsumeIntInRange(-20, 20) for _ in range(numel)]
        return torch.tensor(vals, dtype=dtype).reshape(shape)


def _canonical_dim(dim, rank):
    if rank == 0:
        return 0
    if dim < 0:
        dim += rank
    return dim


@atheris.instrument_func
def TestOneInput(data: bytes):
    fdp = atheris.FuzzedDataProvider(data)
    try:
        rank = fdp.ConsumeIntInRange(1, 4)
        shape = _consume_shape(fdp, rank, min_dim=0, max_dim=4)
        dtype = _pick_dtype(fdp)

        self_tensor = _make_tensor(fdp, shape, dtype)

        raw_dim = fdp.ConsumeIntInRange(-rank, rank - 1)
        dim = raw_dim if rank > 0 else 0
        cdim = _canonical_dim(dim, rank)

        select_shape = list(shape)
        if rank > 0:
            del select_shape[cdim]

        if shape[cdim] == 0:
            index = 0
        else:
            if fdp.ConsumeBool():
                index = fdp.ConsumeIntInRange(0, shape[cdim] - 1)
            else:
                lo = -shape[cdim]
                hi = shape[cdim] - 1
                index = fdp.ConsumeIntInRange(lo, hi)

        if fdp.ConsumeBool():
            src_dtype = dtype
        else:
            src_dtype = _pick_dtype(fdp)

        src = _make_tensor(fdp, select_shape, src_dtype)

        if fdp.ConsumeBool():
            result = torch.select_scatter(self_tensor, src, dim, index)
        else:
            result = torch.select_scatter(input=self_tensor, src=src, dim=dim, index=index)

        if fdp.ConsumeBool():
            _ = result.shape
        if fdp.ConsumeBool():
            _ = result.dtype
        if fdp.ConsumeBool():
            _ = torch.select(result, cdim, index if index >= 0 else shape[cdim] + index) if shape[cdim] > 0 else result

    except (
        RuntimeError,
        TypeError,
        ValueError,
        IndexError,
        AssertionError,
        NotImplementedError,
    ):
        return
    except Exception:
        return


atheris.Setup(sys.argv, TestOneInput)
atheris.Fuzz()
